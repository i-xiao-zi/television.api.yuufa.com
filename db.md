create table public.origins (
  id serial not null,
  title character varying(255) not null default ''::character varying,
  name character varying(255) not null default ''::character varying,
  home character varying(255) null default ''::character varying,
  url character varying(255) not null default ''::character varying,
  sort smallint not null default 99,
  active boolean not null default false,
  crawled_at timestamp without time zone null,
  created_at timestamp without time zone null default now(),
  updated_at timestamp without time zone null,
  deleted_at timestamp without time zone null,
  constraint origins_pkey primary key (id)
) TABLESPACE pg_default;

create table public.errors (
  id serial not null,
  origin_id smallint not null default 99,
  url character varying(255) not null default ''::character varying,
  error text not null default ''::character,
  created_at timestamp without time zone null default now(),
  updated_at timestamp without time zone null,
  deleted_at timestamp without time zone null,
  constraint errors_pkey primary key (id)
) TABLESPACE pg_default;
